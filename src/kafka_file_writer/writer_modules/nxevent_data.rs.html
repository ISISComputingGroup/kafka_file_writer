31impl WriterModule for NXeventdata {
32    fn on_message(
33        &mut self,
34        meta: &KafkaMessageMeta,
35        message: &DeserializedMessage,
36    ) -> Result<(), WriterModuleError> {
37        // TODO: the logic for constructing these datasets from pu00 + ev44 is more complicated
38        // TODO: than this. Additionally, various parts of this need refactoring into separate
39        // TODO: reusable modules.
40        // TODO: Currently this is just a proof-of-principle that writer modules can receive and
41        // TODO: write data.
42        if let Some(timestamp) = meta.timestamp {
43            let timestamp = timestamp.try_into().map_err(|e| WriterModuleError::Fatal {
44                cause: Box::new(e),
45                context: "Kafka message timestamp was negative",
46            })?;
47
48            self.latest_message_time_ms = timestamp;
49            if timestamp < self.run_start_time_ms || timestamp > self.run_stop_time_ms {
50                // This writer module doesn't write messages before / after run.
51                return Ok(());
52            }
53        }
54
55        match message {
56            DeserializedMessage::PulseMetadataPu00(msg) => {
57                let event_time_zero = self
58                    .group
59                    .dataset("event_time_zero")
60                    .err_parent_dataset(&self.group, "event_time_zero")?;
61                let original_length = event_time_zero.size();
62                event_time_zero
63                    .resize(original_length + 1)
64                    .err_dataset(&event_time_zero)?;
65                event_time_zero
66                    .write_slice(&[msg.reference_time() as f64], original_length..)
67                    .err_dataset(&event_time_zero)?;
68            }
69            DeserializedMessage::EventDataEv44(msg) => {
70                let pixel_ids = msg
71                    .pixel_id()
72                    .ok_or_else(|| WriterModuleError::InvalidData("Missing pixel_id".to_owned()))?;
73                let time_of_flight = msg.time_of_flight().ok_or_else(|| {
74                    WriterModuleError::InvalidData("Missing time_of_flight".to_owned())
75                })?;
76
77                let pixel_ids_len = pixel_ids.len();
78                let time_of_flight_len = time_of_flight.len();
79
80                if pixel_ids_len != time_of_flight_len {
81                    return Err(WriterModuleError::InvalidData(format!(
82                        "pixel_id length {} does not match time_of_flight length {}",
83                        pixel_ids_len, time_of_flight_len
84                    )));
85                }
86
87                let event_time_offset = self
88                    .group
89                    .dataset("event_time_offset")
90                    .err_parent_dataset(&self.group, "event_time_offset")?;
91                let event_id = self
92                    .group
93                    .dataset("event_id")
94                    .err_parent_dataset(&self.group, "event_id")?;
95                let original_length = event_time_offset.size();
96                event_id
97                    .resize(original_length + pixel_ids_len)
98                    .err_dataset(&event_id)?;
99                event_time_offset
100                    .resize(original_length + time_of_flight_len)
101                    .err_dataset(&event_time_offset)?;
102
103                event_id
104                    .write_slice(
105                        &pixel_ids.into_iter().collect::<Vec<_>>(),
106                        original_length..,
107                    )
108                    .err_dataset(&event_id)?;
109
110                event_time_offset
111                    .write_slice(
112                        &time_of_flight
113                            .into_iter()
114                            .map(|t| (t as f32) / 1000.)
115                            .collect::<Vec<_>>(),
116                        original_length..,
117                    )
118                    .err_dataset(&event_time_offset)?;
119            }
120            _ => {
121                debug!(
122                    "nxevent_data writer got message unrecognised schema (topic={}, partition={}, offset={})",
123                    meta.topic, meta.partition, meta.offset
124                )
125            }
126        }
127        Ok(())
128    }
129
130    fn on_run_stop(
131        &mut self,
132        _meta: &KafkaMessageMeta,
133        message: &RunStop,
134    ) -> Result<(), WriterModuleError> {
135        self.run_stop_time_ms = message.stop_time();
136        Ok(())
137    }
138
139    fn flush(&mut self) -> Result<(), WriterModuleError> {
140        Ok(())
141    }
142
143    fn finished(&self) -> bool {
144        self.latest_message_time_ms >= self.run_stop_time_ms
145    }
146}

148impl WriterModuleSpec for NXeventdata {
149    type Config = NXeventdataConfig;
150
151    const NAME: &'static str = "NXevent_data";
152
153    fn create(
154        group: hdf5::Group,
155        run_start_parameters: &RunStartParameters,
156        config: Self::Config,
157    ) -> Result<WriterModuleCreateResult<Self>, WriterModuleCreationError> {
158        let subscriptions = vec![Subscription::new(
159            config.topic.to_owned(),
160            SubscriptionKey::All,
161        )];
162
163        let mut module = NXeventdata {
164            _config: config,
165            group,
166            latest_message_time_ms: 0,
167            run_start_time_ms: run_start_parameters.start_time_ms,
168            run_stop_time_ms: run_start_parameters.stop_time_ms,
169        };
170
171        module.init(run_start_parameters)?;
172
173        Ok(WriterModuleCreateResult {
174            module,
175            subscriptions,
176        })
177    }
178}
179
180impl NXeventdata {
181    fn init(
182        &mut self,
183        run_start_parameters: &RunStartParameters,
184    ) -> Result<(), WriterModuleCreationError> {
185        self.run_start_time_ms = run_start_parameters.start_time_ms;
186
187        self.group
188            .new_dataset::<f64>()
189            .shape(SimpleExtents::resizable([0]))
190            .chunk([32000]) // TODO: configurable chunk sizes.
191            .create("event_time_zero")
192            .err_parent_dataset(&self.group, "event_time_zero")?;
193
194        self.group
195            .new_dataset::<u32>()
196            .shape(SimpleExtents::resizable([0]))
197            .chunk([32000]) // TODO: configurable chunk sizes.
198            .create("event_id")
199            .err_parent_dataset(&self.group, "event_id")?;
200
201        self.group
202            .new_dataset::<f32>()
203            .shape(SimpleExtents::resizable([0]))
204            .chunk([32000]) // TODO: configurable chunk sizes.
205            .create("event_time_offset")
206            .err_parent_dataset(&self.group, "event_time_offset")?;
207
208        add_ascii_string_attribute(&self.group, "NX_class", "NXevent_data")?;
209        Ok(())
210    }
