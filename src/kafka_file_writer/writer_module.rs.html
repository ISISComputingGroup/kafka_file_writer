15/// Error type returned by writer modules at runtime.
16#[derive(Error, Debug, Diagnostic)]
17pub enum WriterModuleError {
18    /// Failure to write to a Nexus file.
19    #[error(transparent)]
20    #[diagnostic(transparent)]
21    FailedFileWrite(#[from] Hdf5Error),
22
23    /// Invalid data which the writer module is unable to
24    /// consume. This is non-fatal and the writer module
25    /// can accept new, valid, data in subsequent calls.
26    #[error("Invalid data passed to writer module: {0}")]
27    InvalidData(String),
28
29    /// A generic fatal error from which this writer module
30    /// will not be able to recover. The writer module is
31    /// in a state which can no longer accept new data,
32    /// or in an internally-inconsistent state.
33    #[error("Fatal error in writer module: {context}")]
34    Fatal {
35        #[source]
36        cause: Box<dyn Error + Send + Sync + 'static>,
37        context: &'static str,
38    },
39}

41/// Error type returned by writer modules at configuration time.
42#[derive(Error, Debug, Diagnostic)]
43pub enum WriterModuleCreationError {
44    /// The writer-module configuration is semantically invalid
45    /// or inconsistent, and cannot be used.
46    #[error("Invalid writer module configuration: {}", description)]
47    ConfigInvalid { description: String },
48
49    /// The configuration of this writer module failed to parse
50    /// into the expected structure from JSON.
51    #[error("Writer module config parse of type '{}' failed", config_type_name)]
52    ConfigParseError {
53        #[source]
54        cause: serde_json::Error,
55
56        config_type_name: &'static str,
57    },
58
59    #[error(transparent)]
60    #[diagnostic(transparent)]
61    InitializationHdf5Error(#[from] Hdf5Error),
62}

64/// Result of calling `create` on a writer module spec.
65pub struct WriterModuleCreateResult<T> {
66    /// The `Writer` implementation
67    pub module: T,
68    /// Subscriptions defining which Kafka streams this writer
69    /// needs to listen to.
70    pub subscriptions: Vec<Subscription>,
71}

73/// Runtime behaviour definitions for this writer module type.
74pub trait WriterModule: Debug {
75    /// Called for each received message which matched a ``Subscription``
76    /// provided by this module.
77    ///
78    /// The implementation of this method may write directly to the NeXus file,
79    /// or buffer data in-memory.
80    fn on_message(
81        &mut self,
82        meta: &KafkaMessageMeta,
83        message: &DeserializedMessage,
84    ) -> Result<(), WriterModuleError>;
85
86    /// Called when a run-stop message is received.
87    ///
88    /// The implementation of this method may write directly to the NeXus file,
89    /// or buffer data in-memory.
90    fn on_run_stop(
91        &mut self,
92        meta: &KafkaMessageMeta,
93        message: &RunStop,
94    ) -> Result<(), WriterModuleError>;
95
96    /// Ensure that any local state is reflected in the NeXus file, in an
97    /// internally-consistent way.
98    ///
99    /// For example, this may write an in-memory histogram to the file,
100    /// or write in-memory summary statistics such as min/max/average.
101    ///
102    /// After this method completes, the file must be in a self-consistent
103    /// state and as up-to-date as possible with data received so far.
104    ///
105    /// This may be implemented as a no-op if this writer module always streams
106    /// all data to file directly and does not buffer values in-memory.
107    ///
108    /// This may be called multiple times during a single run, for example
109    /// to generate an autosave/intermediate file and then after run-end.
110    fn flush(&mut self) -> Result<(), WriterModuleError>;
111
112    /// Has this writer module finished writing, after a run-stop was received?
113    ///
114    /// Due to out-of-order Kafka message delivery, a run stop may be received
115    /// before messages which occured during the run. This hook allows a writer
116    /// module to declare whether it has received all necessary data up to
117    /// run end. It should return `true` if all necessary data has been received,
118    /// or `false` otherwise.
119    fn finished(&self) -> bool;
120}

122/// Creation parameters for writer modules of this type. This trait
123/// should be implemented by all writer modules, and is necessary
124/// in order to add the writer module to the ``WriterModuleRegistry``.
125pub trait WriterModuleSpec: WriterModule + Sized {
126    /// The configuration parameters for this type.
127    /// This type will be automatically deserialized from the JSON parameters
128    /// specified in `nexus_structure`.
129    type Config: DeserializeOwned;
130
131    /// The name of this writer module, as referred to in `nexus_structure`.
132    /// This must be unique among all writer modules.
133    const NAME: &'static str;
134
135    /// Create the writer module.
136    ///
137    /// This should return ``Err(WriterModuleFactoryError::ConfigInvalid)``
138    /// if the configuration is semantically invalid or inconsistent.
139    fn create(
140        parent_group: hdf5::Group,
141        run_start_parameters: &RunStartParameters,
142        config: Self::Config,
143    ) -> Result<WriterModuleCreateResult<Self>, WriterModuleCreationError>;
144}
