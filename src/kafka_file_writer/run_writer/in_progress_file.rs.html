18/// Represents a file which is in the process of being written to.
19pub struct InProgressFile<'a> {
20    file: hdf5::File,
21    writers: Vec<PlacedWriter>,
22    run_start_parameters: &'a RunStartParameters,
23    run_stop_received: bool,
24    router: MessageRouter,
25}
26
27impl<'a> InProgressFile<'a> {
28    pub fn new(
29        file: hdf5::File,
30        run_start_parameters: &'a RunStartParameters,
31        structure: NexusFileStructure,
32        factories: &WriterModuleFactories,
33    ) -> Result<Self, FileWriterError> {
34        let mut result = InProgressFile {
35            file,
36            writers: vec![],
37            run_start_parameters,
38            router: MessageRouter::default(),
39            run_stop_received: false,
40        };
41
42        result.use_structure(structure, factories)?;
43        Ok(result)
44    }
45
46    fn use_structure(
47        &mut self,
48        structure: NexusFileStructure,
49        factories: &WriterModuleFactories,
50    ) -> Result<(), FileWriterError> {
51        let root = self.file.clone();
52        for child in structure.children {
53            self.add_child(&root, child, factories)?;
54        }
55        for (key, json_value) in structure.attributes {
56            if let Value::String(value) = json_value {
57                add_ascii_string_attribute(&root, &key, &value)?;
58            }
59        }
60        add_root_dataset_attributes(&root)?;
61        Ok(())
62    }
63
64    fn add_child(
65        &mut self,
66        parent: &hdf5::Group,
67        structure: NexusStructureItem,
68        factories: &WriterModuleFactories,
69    ) -> Result<(), FileWriterError> {
70        match structure {
71            NexusStructureItem::Group(group) => {
72                self.add_nexus_group(parent.clone(), group, factories)
73            }
74            NexusStructureItem::WriterModule(module_spec) => {
75                self.add_placed_writer(PlacedWriter::new(
76                    parent.clone(),
77                    module_spec,
78                    self.run_start_parameters,
79                    factories,
80                )?);
81                Ok(())
82            }
83            NexusStructureItem::Invalid(json) => Err(FileWriterError::NexusStructureError(
84                NexusStructureError::UnknownChildType {
85                    location: parent.name(),
86                    invalid_json: json.to_string(),
87                },
88            )),
89        }
90    }
91
92    fn add_nexus_group(
93        &mut self,
94        parent: hdf5::Group,
95        group: NexusGroup,
96        writer_module_factories: &WriterModuleFactories,
97    ) -> Result<(), FileWriterError> {
98        trace!(
99            "Creating new group {} with parent {}",
100            group.name,
101            parent.name()
102        );
103        let hdf_group = parent
104            .create_group(&group.name)
105            .err_parent_dataset(&parent, &group.name)?;
106
107        for (key, json_value) in group.attributes {
108            if let Value::String(value) = json_value {
109                add_ascii_string_attribute(&hdf_group, &key, &value)?;
110            }
111        }
112
113        for child in group.children {
114            self.add_child(&hdf_group, child, writer_module_factories)?;
115        }
116        Ok(())
117    }
118
119    fn add_placed_writer(&mut self, writer: PlacedWriter) {
120        let idx = self.writers.len();
121
122        for sub in writer.subscriptions().iter() {
123            self.router.subscribe(idx, sub);
124        }
125
126        self.writers.push(writer);
127    }
128
129    pub fn topics(&self) -> impl Iterator<Item = &str> {
130        self.router.topics()
131    }
132
133    pub fn job_id(&self) -> &str {
134        &self.run_start_parameters.job_id
135    }
136
137    pub fn finished(&self) -> bool {
138        self.run_stop_received && self.writers.iter().all(|w| w.writer().finished())
139    }
140
141    pub fn on_message(
142        &mut self,
143        meta: &KafkaMessageMeta,
144        msg: &DeserializedMessage,
145    ) -> Result<(), FileWriterError> {
146        for writer_index in self.router.writers_for(meta) {
147            if let Some(writer) = self.writers.get_mut(writer_index) {
148                writer.on_message(meta, msg)?;
149            } else {
150                error!(
151                    "Writer module indices and subscriptions are out of sync; no writer module for idx={}",
152                    writer_index
153                );
154                return Err(FileWriterError::WriterModuleNotFound { writer_index });
155            }
156        }
157        Ok(())
158    }
159
160    pub fn on_run_stop(
161        &mut self,
162        meta: &KafkaMessageMeta,
163        run_stop: &RunStop,
164    ) -> Result<(), FileWriterError> {
165        for writer in self.writers.iter_mut() {
166            writer.on_run_stop(meta, run_stop)?;
167        }
168        self.run_stop_received = true;
169        Ok(())
170    }
171}

268    fn test_in_progress_file_with_empty_structure() {
269        with_nexus_file(FileCreator::Memory, "test", |f| {
270            let rsp = RunStartParameters::default();
271            let ipf = InProgressFile::new(f, &rsp, empty_structure(), &default_registry()).unwrap();
272            assert_eq!(ipf.topics().collect::<Vec<&str>>(), Vec::<&str>::new());
273            assert_eq!(ipf.job_id(), rsp.job_id);

275            Ok(())
