28pub struct SingleFileWriteTask<'a> {
29    config: &'a GlobalConfig,
30    writer_module_factories: &'a WriterModuleFactories,
31    run_start_parameters: &'a RunStartParameters,
32    run_end_tracker: RunEndTracker<'a>,
33}
34
35impl<'a> SingleFileWriteTask<'a> {
36    pub fn new(
37        config: &'a GlobalConfig,
38        writer_module_factories: &'a WriterModuleFactories,
39        run_start_parameters: &'a RunStartParameters,
40    ) -> Self {
41        let run_end_tracker = RunEndTracker::new(config);
42
43        SingleFileWriteTask {
44            config,
45            writer_module_factories,
46            run_start_parameters,
47            run_end_tracker,
48        }
49    }
50
51    pub fn write_data_for_run(
52        &mut self,
53        consumer: &impl Stream,
54        file_factory: FileCreator,
55    ) -> Result<(), FileWriterError> {
56        let resolved_structure = self.run_start_parameters.structure(self.config)?;
57
58        let mut path = self.config.file_output_directory.clone();
59        path.push(&self.run_start_parameters.filename);
60
61        with_nexus_file(file_factory, &path, |file| {
62            let mut in_progress_file = InProgressFile::new(
63                file,
64                self.run_start_parameters,
65                resolved_structure,
66                self.writer_module_factories,
67            )?;
68
69            let seek_timestamp: i64 = self
70                .run_start_parameters
71                .start_time_ms
72                .saturating_sub(self.config.pre_run_seek_time_ms)
73                .try_into()
74                .map_err(|e| FileWriterError::IntegerError {
75                    cause: e,
76                    context: "timestamp greater than i64::MAX in seek timestamp",
77                })?;
78
79            let topics = in_progress_file
80                .topics()
81                .map(|s| s.to_string())
82                .chain(once(
83                    self.run_start_parameters
84                        .control_topic(self.config)
85                        .to_owned(),
86                ))
87                .unique()
88                .collect::<Vec<_>>();
89
90            with_assignment(
91                consumer,
92                &topics.iter().map(|s| s.as_str()).collect::<Vec<_>>(),
93                seek_timestamp,
94                Duration::from_millis(self.config.kafka_assignment_timeout_ms),
95                || {
96                    self.handle_messages_until_run_end(
97                        consumer,
98                        seek_timestamp,
99                        &mut in_progress_file,
100                    )
101                },
102            )
103        })
104    }
105
106    fn handle_messages_until_run_end(
107        &mut self,
108        consumer: &impl Stream,
109        seek_timestamp: i64,
110        in_progress_file: &mut InProgressFile,
111    ) -> Result<(), FileWriterError> {
112        // Initialize the offsets we've 'consumed' to the offsets corresponding to our seek time.
113        loop {
114            match consumer.offsets_for_timestamp(
115                seek_timestamp,
116                Duration::from_millis(self.config.kafka_metadata_timeout_ms),
117            ) {
118                Ok(offsets) => {
119                    self.run_end_tracker.set_initial_offsets(offsets);
120                    break;
121                }
122                Err(e) => {
123                    let report = Report::from(e);
124                    warn!(
125                        "Error while getting initial offsets (will retry): {:?}",
126                        report
127                    );
128                    thread::sleep(Duration::from_millis(self.config.stream_error_backoff_ms));
129                }
130            }
131        }
132
133        if let Some(stop_time) = self.run_start_parameters.stop_time_ms {
134            self.run_end_tracker.on_run_stop(stop_time, consumer);
135        }
136
137        loop {
138            match consumer.poll(Duration::from_millis(
139                self.config.data_consumer_poll_time_ms,
140            )) {
141                Some(Ok(msg)) => {
142                    debug!(
143                        "Handling message on topic='{}', partition={}, offset={}",
144                        msg.topic(),
145                        msg.partition(),
146                        msg.offset()
147                    );
148
149                    self.handle_one_message(&msg, in_progress_file, consumer)?;
150                }
151                Some(Err(e)) => {
152                    let report = Report::from(e);
153                    warn!("Error during consumer poll (will retry): {:?}", report);
154                    counter!(KAFKA_POLL_ERRORS).increment(1);
155                    thread::sleep(Duration::from_millis(self.config.stream_error_backoff_ms));
156                }
157                None => {}
158            }
159
160            match self.run_end_tracker.poll_complete(consumer) {
161                Ok(true) => {
162                    break;
163                }
164                Ok(false) => {}
165                Err(e) => {
166                    let report = Report::from(e);
167                    warn!(
168                        "Error while polling for run completion (will retry): {:?}",
169                        report
170                    );
171                    thread::sleep(Duration::from_millis(self.config.stream_error_backoff_ms));
172                }
173            }
174        }
175
176        in_progress_file.flush()?;
177
178        Ok(())
179    }
180
181    fn handle_one_message(
182        &mut self,
183        msg: &impl Message,
184        in_progress_file: &mut InProgressFile,
185        consumer: &impl Stream,
186    ) -> Result<(), FileWriterError> {
187        let meta = KafkaMessageMeta {
188            topic: msg.topic(),
189            key: msg.key(),
190            timestamp: msg.timestamp().to_millis(),
191            partition: msg.partition(),
192            offset: msg.offset(),
193        };
194
195        self.run_end_tracker.message_consumed(&meta);
196
197        if let Some(payload) = msg.payload() {
198            counter!(INCOMING_DATA_MESSAGES).increment(1);
199            counter!(INCOMING_DATA_MESSAGE_SIZE).increment(payload.len() as u64);
200
201            let start_time = Instant::now();
202
203            let parsed_msg = deserialize_message(payload);
204            match parsed_msg {
205                Ok(DeserializedMessage::RunStop6s4t(run_stop)) => {
206                    self.handle_stop(&meta, &run_stop, in_progress_file, consumer)?;
207                }
208                Ok(data) => in_progress_file.on_message(&meta, &data)?,
209                Err(err) => {
210                    warn!(
211                        "Ignoring data message on topic '{}' (partition={}, offset={}, length={:?} bytes) as failed to deserialize\nCaused by: {:?}",
212                        msg.topic(),
213                        msg.partition(),
214                        msg.offset(),
215                        msg.payload().map(|p| p.len()),
216                        err
217                    );
218                    counter!(INCOMING_DATA_MESSAGES_FAILED_DESERIALIZE).increment(1);
219                }
220            }
221            if let Some(schema_bytes) = get_schema_id(payload)
222                && let Ok(schema) = str::from_utf8(schema_bytes)
223            {
224                histogram!(SCHEMA_PROCESSING_TIMES, "schema" => schema.to_owned())
225                    .record(start_time.elapsed().as_secs_f64());
226            }
227        } else {
228            warn!(
229                "Message on topic='{}', partition={}, offset={} had no payload; ignoring",
230                msg.topic(),
231                msg.partition(),
232                msg.offset()
233            );
234            counter!(INCOMING_DATA_MESSAGES_NO_PAYLOAD).increment(1);
235        }
236
237        Ok(())
238    }
239
240    fn handle_stop(
241        &mut self,
242        meta: &KafkaMessageMeta,
243        run_stop: &RunStop,
244        in_progress_file: &mut InProgressFile,
245        consumer: &impl Stream,
246    ) -> Result<(), FileWriterError> {
247        if run_stop.job_id() != Some(in_progress_file.job_id()) {
248            debug!(
249                "Ignoring run stop with mismatched job id (got '{:?}', currently writing '{}')",
250                run_stop.job_id(),
251                in_progress_file.job_id()
252            );
253            return Ok(());
254        }
255        info!(
256            "Accepted run-stop for Job ID '{}'",
257            in_progress_file.job_id()
258        );
259        in_progress_file.on_run_stop(meta, run_stop)?;
260        self.run_end_tracker
261            .on_run_stop(run_stop.stop_time(), consumer);
262        Ok(())
263    }

272    #[test]
273    fn test_write_data_for_run() {
274        let config = GlobalConfig::test_config();
275
276        let rsp = RunStartParameters {
277            start_time_ms: 123,
278            stop_time_ms: None,
279            job_id: FakeStream::TEST_JOB_ID.to_string(),
280            nexus_structure: FakeStream::TEST_STRUCTURE.to_string(),
281            control_topic: None,
282            filename: FakeStream::TEST_FILENAME.into(),
283            n_periods: 1,
284            metadata: None,
285            instrument_name: "UNITTEST".to_owned(),
286        };
287
288        let consumer = FakeStream::default();
289
290        consumer.append_runinfo_message_with_payload(Some(b"nonsense".to_vec()));
291        consumer.append_runinfo_message_with_payload(None);
292        // Append two runstops; only the first should be consumed, after the two messages
293        // above are ignored due to being invalid (these should not cause an `Err`, just
294        // a warning).
295        consumer.append_runinfo_run_stop_message();
296        consumer.append_runinfo_run_stop_message();
297
298        let reg = default_registry();
299
300        let mut task = SingleFileWriteTask::new(&config, &reg, &rsp);
301
302        task.write_data_for_run(&consumer, FileCreator::Memory)
303            .unwrap();
304
305        assert_eq!(consumer.messages.borrow().len(), 1);
306    }
