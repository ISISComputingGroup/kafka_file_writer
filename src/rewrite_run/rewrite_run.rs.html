1//! Rewrite a NeXus file from data in Kafka.
2//!
3//! This utility writes a single run *as though* a run start with the specified command-line
4//! parameters was seen; it does not actually emit a run start to Kafka.
5//!
6//! Because it does not depend on actually seeing a run start or run-stop in Kafka, this utility
7//! is suitable for:
8//! - Re-writing data from Kafka when the original run-start or run-stop was missing
9//! - Testing writing real data from a beamline without needing that beamline to start and stop data acquisition.
10//!
11//! A full example command is:
12//! ```
13//! rewrite_run --config config.toml --start-time-ms 1794307704000 --stop-time-ms 1794307724000 -f hrpd.nxs -i hrpd -s structures\isis.json
14//! ```
15
16use clap::Parser;
17use kafka_file_writer::config::GlobalConfig;
18use kafka_file_writer::hdf::file_creator::FileCreator;
19use kafka_file_writer::metrics::initialize_metrics;
20use kafka_file_writer::metrics::kafka::MetricsKafkaContext;
21use kafka_file_writer::run_writer::run_start_parameters::RunStartParameters;
22use kafka_file_writer::run_writer::single_file_write_task::SingleFileWriteTask;
23use kafka_file_writer::stream::kafka::{KafkaStream, make_data_kafka_client_config};
24use kafka_file_writer::writer_modules::default_registry;
25use miette::{IntoDiagnostic, Result};
26use std::path::PathBuf;
27use uuid::Uuid;
28
29#[derive(Parser, Debug)]
30#[command(version, about, long_about = None)]
31struct Args {
32    /// Path to config file
33    #[arg(short = 'c', long)]
34    config: PathBuf,
35
36    /// Run start time, as milliseconds since the Unix epoch.
37    #[arg(long)]
38    start_time_ms: u64,
39
40    /// Run stop time, as milliseconds since the Unix epoch.
41    #[arg(long)]
42    stop_time_ms: u64,
43
44    /// Path to the NeXus file to be written.
45    ///
46    /// If relative, the path will be relative to the `file_output_directory` specified in
47    /// the config. If absolute, the file will be written in the specified location.
48    #[arg(short = 'f', long)]
49    filename: PathBuf,
50
51    /// Path to a nexus-structure JSON (for example, one of the structures in the
52    /// `structures` directory).
53    #[arg(long, short = 's')]
54    nexus_structure_file: PathBuf,
55
56    /// Number of periods in this run.
57    #[arg(short = 'p', long, default_value = "1")]
58    n_periods: u32,
59
60    /// Instrument name for this run. This is written into the resulting NeXus file,
61    /// and may be used by analysis programs.
62    #[arg(long, short = 'i')]
63    instrument_name: String,
64
65    /// Serve metrics for this process if specified
66    #[arg(long, action)]
67    metrics: bool,
68
69    #[command(flatten)]
70    verbosity: clap_verbosity_flag::Verbosity,
71}
72
73fn main() -> Result<()> {
74    let args = Args::try_parse().into_diagnostic()?;
75
76    env_logger::Builder::new()
77        .filter_level(args.verbosity.into())
78        .format_timestamp_micros()
79        .init();
80
81    let config = GlobalConfig::from_path(&args.config)?;
82
83    if args.metrics {
84        initialize_metrics(&config).into_diagnostic()?;
85    }
86
87    let stream = KafkaStream::from_config(
88        &make_data_kafka_client_config(&config),
89        MetricsKafkaContext { name: "data" },
90    )?;
91
92    let nexus_structure = std::fs::read_to_string(args.nexus_structure_file).into_diagnostic()?;
93
94    let parameters = RunStartParameters {
95        filename: args.filename,
96        instrument_name: args.instrument_name,
97        job_id: Uuid::new_v4().to_string(),
98        start_time_ms: args.start_time_ms,
99        stop_time_ms: Some(args.stop_time_ms),
100        nexus_structure,
101        metadata: None,
102        n_periods: args.n_periods,
103        control_topic: None,
104    };
105
106    let writer_module_registry = default_registry();
107
108    let mut task = SingleFileWriteTask::new(&config, &writer_module_registry, &parameters);
109    task.write_data_for_run(&stream, FileCreator::Disk)?;
110
111    Ok(())
112}
